   468→
   469→async function doInit(): Promise<void> {
   470→  if (process.env.SKIP_DB_INIT === "true") return;
   471→
   472→  console.log("[db-init] Starting runtime database initialization…");
   473→  const db = new PrismaClient();
   474→
   475→  try {
   476→    // 1. Create tables if missing (raw SQL — no prisma CLI needed)
   477→    console.log("[db-init] Creating tables if missing…");
   478→    for (const sql of CREATE_TABLES_SQL) {
   479→      try {
   480→        await db.$executeRawUnsafe(sql);
   481→      } catch (err: any) {
   482→        // Ignore "already exists" errors
   483→        if (!/already exists|Duplicate/i.test(err.message)) {
   484→          console.warn(`[db-init] Table create warning: ${err.message.slice(0, 200)}`);
   485→        }
   486→      }
   487→    }
   488→    console.log("[db-init] ✓ Tables ready");
   489→
   490→    // 2. Seed admin user
   491→    const adminEmail = (process.env.ADMIN_EMAIL ?? "admin@allcombiner.com").toLowerCase();
   492→    const adminPassword = process.env.ADMIN_PASSWORD ?? "Admin123!2024";
   493→    const adminName = process.env.ADMIN_NAME ?? "Admin";
   494→
   495→    const existingAdmin = await db.user.findUnique({ where: { email: adminEmail } }).catch(() => null);
   496→    if (!existingAdmin) {
   497→      const passwordHash = await bcrypt.hash(adminPassword, 12);
   498→      const id = `admin_${Date.now()}_${Math.random().toString(36).slice(2, 10)}`;
   499→      await db.user.create({
   500→        data: { id, email: adminEmail, name: adminName, passwordHash, role: "ADMIN", credits: 9999, updatedAt: new Date() },
   501→      }).catch((e: any) => console.warn(`[db-init] Admin create failed: ${e.message}`));
   502→      console.log(`[db-init] ✓ Admin created: ${adminEmail} (credits: 9999)`);
   503→    } else if (existingAdmin.role !== "ADMIN") {
   504→      await db.user.update({ where: { id: existingAdmin.id }, data: { role: "ADMIN", credits: 9999, updatedAt: new Date() } }).catch(() => {});
   505→      console.log(`[db-init] ✓ User promoted to admin: ${adminEmail} (credits: 9999)`);
   506→    } else {
   507→      // Ensure admin always has 9999 credits (in case it was changed)
   508→      if (existingAdmin.credits !== 9999) {
   509→        await db.user.update({ where: { id: existingAdmin.id }, data: { credits: 9999, updatedAt: new Date() } }).catch(() => {});
   510→      }
   511→      console.log("[db-init] ✓ Admin already exists");
   512→    }
   513→
   514→    // 3. Seed AI models
   515→    for (const m of DEFAULT_MODELS) {
   516→      const existing = await db.aIModel.findFirst({ where: { providerId: m.providerId, name: m.name } }).catch(() => null);
   517→      if (!existing) {
   518→        const id = `model_${Date.now()}_${Math.random().toString(36).slice(2, 10)}`;
   519→        await db.aIModel.create({
   520→          data: {
   521→            id,
   522→            name: m.name,
   523→            providerId: m.providerId,
   524→            provider: m.provider,
   525→            description: m.description,
   526→            costPerCall: m.costPerCall,
   527→            creditCost: m.creditCost,
   528→            enabled: Boolean(m.enabled),
   529→            isActive: Boolean(m.isActive),
   530→            updatedAt: new Date(),
   531→          },
   532→        }).catch((e: any) => console.warn(`[db-init] Model create failed: ${e.message}`));
   533→        console.log(`[db-init] ✓ Model created: ${m.name} (creditCost: ${m.creditCost})`);
   534→      }
   535→    }
   536→
   537→    // Ensure exactly one active model
   538→    const activeCount = await db.aIModel.count({ where: { isActive: true } }).catch(() => 0);
   539→    if (activeCount === 0) {
   540→      const first = await db.aIModel.findFirst({ orderBy: { createdAt: "asc" } }).catch(() => null);
   541→      if (first) {
   542→        await db.aIModel.update({ where: { id: first.id }, data: { isActive: true, updatedAt: new Date() } }).catch(() => {});
   543→        console.log(`[db-init] ✓ Activated default model: ${first.name}`);
   544→      }
   545→    }
   546→
   547→    // 4. Seed pricing plans
   548→    for (const p of DEFAULT_PLANS) {
   549→      const existing = await db.pricingPlan.findUnique({ where: { slug: p.slug } }).catch(() => null);
   550→      if (!existing) {
   551→        const id = `plan_${Date.now()}_${Math.random().toString(36).slice(2, 10)}`;
   552→        await db.pricingPlan.create({
   553→          data: {
   554→            id,
   555→            slug: p.slug,
   556→            nameJson: p.nameJson,
   557→            description: p.description,
   558→            priceMonthly: p.priceMonthly,
   559→            priceYearly: p.priceYearly,
   560→            currency: p.currency || "EUR",
   561→            credits: p.credits,
   562→            featured: Boolean(p.featured),
   563→            enabled: Boolean(p.enabled),
   564→            sortOrder: p.sortOrder,
   565→            whopPlanId: p.whopPlanId ?? null,
   566→            whopCheckoutUrl: p.whopCheckoutUrl ?? null,
   567→            billingPeriod: p.billingPeriod ?? null,
   568→            updatedAt: new Date(),
   569→          },
   570→        }).catch((e: any) => console.warn(`[db-init] Plan create failed: ${e.message}`));
   571→        console.log(`[db-init] ✓ Plan created: ${p.slug}${p.whopPlanId ? ` (whop: ${p.whopPlanId})` : ""}`);
   572→      } else {
   573→        // Update existing plan: sync ALL fields from DEFAULT_PLANS so the
   574→        // admin can change them later via the UI, but the seed always
   575→        // reflects the latest known configuration.
   576→        const needsUpdate =
   577→          (p.whopPlanId !== undefined && p.whopPlanId !== existing.whopPlanId) ||
   578→          (p.whopCheckoutUrl !== undefined && p.whopCheckoutUrl !== existing.whopCheckoutUrl) ||
   579→          (p.currency && existing.currency !== p.currency) ||
   580→          (p.billingPeriod && existing.billingPeriod !== p.billingPeriod) ||
   581→          existing.priceMonthly !== p.priceMonthly ||
   582→          existing.credits !== p.credits ||
   583→          existing.description !== p.description;
   584→        if (needsUpdate) {
   585→          await db.pricingPlan.update({
   586→            where: { id: existing.id },
   587→            data: {
   588→              ...(p.whopPlanId !== undefined ? { whopPlanId: p.whopPlanId } : {}),
   589→              ...(p.whopCheckoutUrl !== undefined ? { whopCheckoutUrl: p.whopCheckoutUrl } : {}),
   590→              ...(p.currency ? { currency: p.currency } : {}),
   591→              ...(p.billingPeriod ? { billingPeriod: p.billingPeriod } : {}),
   592→              priceMonthly: p.priceMonthly,
   593→              credits: p.credits,
   594→              description: p.description,
   595→              updatedAt: new Date(),
   596→            },
   597→          }).catch(() => {});
   598→          console.log(`[db-init] ✓ Plan updated: ${p.slug} (synced config from seed)`);
   599→        }
   600→      }
   601→    }
   602→
   603→    // === Cleanup: disable legacy plans (without sub_/recharge_ prefix) ===
   604→    // These are leftover from the original 4-plan grid (free/starter/pro/business).
   605→    // They were replaced by the new slugs (sub_free/sub_starter/etc.).
   606→    // We disable them so they don't appear in the pricing page or admin list.
   607→    const legacySlugs = ["free", "starter", "pro", "business"];
   608→    for (const slug of legacySlugs) {
   609→      const legacy = await db.pricingPlan.findUnique({ where: { slug } }).catch(() => null);
   610→      if (legacy && legacy.enabled) {
   611→        await db.pricingPlan.update({
   612→          where: { id: legacy.id },
   613→          data: { enabled: false, updatedAt: new Date() },
   614→        }).catch(() => {});
   615→        console.log(`[db-init] ✓ Disabled legacy plan: ${slug} (replaced by sub_${slug})`);
   616→      }
   617→    }
   618→
   619→    // 5. Seed fusion prompt
   620→    const existingPrompt = await db.siteSettings.findUnique({ where: { key: "fusion_prompt" } }).catch(() => null);
   621→    if (!existingPrompt) {
   622→      const id = `setting_${Date.now()}_${Math.random().toString(36).slice(2, 10)}`;
   623→      await db.siteSettings.create({
   624→        data: {
   625→          id,
   626→          key: "fusion_prompt",
   627→          value: DEFAULT_PROMPT,
   628→          description: "Central prompt sent to the AI model with the two reference images.",
   629→          updatedAt: new Date(),
   630→        },
   631→      }).catch((e: any) => console.warn(`[db-init] Prompt create failed: ${e.message}`));
   632→      console.log("[db-init] ✓ Fusion prompt seeded");
   633→    }
   634→
   635→    console.log("[db-init] 🎉 Database initialization complete");
   636→  } catch (err) {
   637→    console.error("[db-init] ❌ Initialization failed:", err);
   638→    // Don't throw — let the server start anyway so admin can debug via UI
   639→  } finally {
   640→    await db.$disconnect();
   641→  }
   642→}
   643→
   644→export function ensureDbInitialized(): Promise<void> {
   645→  if (!initPromise) {
   646→    initPromise = doInit();
   647→  }
   648→  return initPromise;
   649→}
